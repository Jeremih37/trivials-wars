   587→/**
   588→ * Genera un array de `n` índices [0..n-1] barajados con Fisher-Yates.
   589→ * Garantiza que cada vez que se llama da una secuencia distinta,
   590→ * para que los datos curiosos aparezcan siempre en orden aleatorio.
   591→ */
   592→function shuffleIds(n: number): number[] {
   593→  const arr = Array.from({ length: n }, (_, i) => i)
   594→  for (let i = n - 1; i > 0; i--) {
   595→    const j = Math.floor(Math.random() * (i + 1))
   596→    ;[arr[i], arr[j]] = [arr[j], arr[i]]
   597→  }
   598→  return arr
   599→}
